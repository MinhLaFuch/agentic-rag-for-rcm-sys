def test_save_load_corpus_roundtrip():
    """Test that save and load produce identical results."""
    # Create a simple corpus
    meta = pd.DataFrame({
        "parent_asin": ["a", "b", "c"],
        "title": ["Item A", "Item B", "Item C"],
        "domain": ["Test", "Test", "Test"],
        "price": [10.0, 20.0, 30.0],
        "categories": [["cat1", "cat2"], ["cat1"], ["cat3"]],

    with tempfile.TemporaryDirectory() as tmpdir:
        corpus_path = Path(tmpdir) / "test.sqlite"
        meta_path = Path(tmpdir) / "meta_test.jsonl.gz"

        # Verify item_domains
