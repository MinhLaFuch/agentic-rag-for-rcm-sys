    # Fit model
    model = ItemKNNRecommender(k).fit(fit_pairs)

    # Save

        path = Path(tmpdir) / "model.npz"
        save_item_knn(model, path, k, fit_pairs)

        # Load

        # Compare similarity matrices

        )
        assert np.array_equal(

        )
