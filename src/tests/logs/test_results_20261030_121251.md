def test_session_reference_is_resolved(corpus):

    from package.agents.session import Session

    session = Session()
    session.add_shown_items([VG_A, VG_B])

    # Use a reference to the second shown item
    executor = PlanExecutor(
        ScriptedLLM(plan({"tool": "QueryTool", "args": {"item_ids": "$S.shown_items.1"}})),
        [QueryTool(corpus)],
        session=session
    )
    results = executor.run("tell me about the second item")
    # Check the result

    assert VG_B in results[0]["args"]["item_ids"]
