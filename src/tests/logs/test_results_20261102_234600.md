def test_quality_gate_rejects_worse_plan():
    """Reflected plan with lower quality is rejected."""
    # Plan 1: SQLTool + RecoModelTool (quality = 0.7)
    sql_tool = FakeSQLTool(fixed_data={"candidates": build_items("item1"), "total_matches": 1, "truncated": False, "applied_filters": {}})
    reco_tool = FakeRecoModelTool()

    plan1 = build_plan_json(
        {"tool": "SQLTool", "args": {"filters": {"domain": "Video_Games"}}},
        {"tool": "RecoModelTool", "args": {"user_id": "u1", "candidates": "$1.candidates", "top_k": 5}},
    )
    # Plan 2: only SQLTool (quality = 0.5) - worse
    plan2 = build_plan_json({"tool": "SQLTool", "args": {"filters": {"domain": "Video_Games"}}})

    llm = ScriptedLLM(plan1, plan2)
    # Set threshold to 0.8 - triggers reflection on plan1 (0.7 < 0.8), then rejects plan2 (0.5 < 0.7)
    executor = PlanExecutor(llm, [sql_tool, reco_tool], reflect_rounds=1, reflect_quality_threshold=0.8)
    results = executor.run("test")

    # Reflection should occur but keep the better original plan
    assert executor.last_reflections == 1
    # Should keep plan 1 results (2 steps) since plan2 is worse
    assert len(results) == 2
    assert results[0]["ok"]
    assert results[1]["ok"]
