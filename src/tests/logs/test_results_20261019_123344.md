```python
def test_example_a_personalized_with_memorytool(corpus, cf_setup):
    """Test example (a): MemoryTool -> SQLTool -> RecoModelTool -> QueryTool."""
    from package.agents import build_tools
    from package.tools import ItemCFTool, RecoModelTool, SemanticSearchTool

    interactions, user2id, item2id = cf_setup
    knn = ItemKNNRecommender(5).fit(interactions)
    tools = build_tools(knn, BaselineScorer(knn), user2id, item2id, interactions, corpus)

    # Example (a): MemoryTool -> SQLTool -> RecoModelTool -> QueryTool
    steps = [
        {"tool": "MemoryTool", "args": {"user_id": "u1"}},
        {"tool": "SQLTool", "args": {"filters": {"categories_any": "$1.top_categories", "price_max": "$1.price_range.max"}, "limit": 30}},
        {"tool": "RecoModelTool", "args": {"user_id": "u1", "candidates": "$2.candidates", "top_k": 5}},
        {"tool": "QueryTool", "args": {"item_ids": "$3.ranked.*.item_id"}},
    ]

    results = PlanExecutor(ScriptedLLM(plan(*steps)), list(tools.for_planner().values())).run("personalized recommendations")
    assert results[0]["ok"]  # MemoryTool
    assert results[1]["ok"]  # SQLTool
    assert results[2]["ok"]  # RecoModelTool
    assert results[3]["ok"]  # QueryTool

    # Verify references resolved correctly
    assert results[1]["args"]["categories_any"] == results[0]["data"]["top_categories"]
    assert results[2]["args"]["candidates"] == results[1]["data"]["candidates"]
    assert len(results[3]["args"]["item_ids"]) == len(results[2]["data"]["ranked"])
```
