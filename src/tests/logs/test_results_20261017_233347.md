def test_relative_imports_resolve():
    """Ensure all relative imports in package/ resolve to existing modules."""
    offenders = []

    for path in (PROJECT_ROOT / "package").rglob("*.py"):
        if "__pycache__" in path.parts:
            continue

        try:
            tree = ast.parse(path.read_text(encoding="utf-8"))

            for node in ast.walk(tree):
                if isinstance(node, ast.ImportFrom) and node.level > 0:
                    # This is a relative import
                    # Calculate the absolute module name
                    parts = list(path.relative_to(PROJECT_ROOT / "package").with_suffix("").parts)
                    depth = node.level
                    if depth > 0:
                        parts = parts[:-depth]

                    module_base = ".".join(parts) if parts else "package"
                    module = node.module or ""
                    full_name = f"{module_base}.{module}" if module else module_base

                    if not _module_exists(full_name):
                        rel_path = path.relative_to(PROJECT_ROOT)
                        offenders.append(f"{rel_path}:{node.lineno}: from {'.' * node.level}{node.module or ''}")
        except SyntaxError:
            continue

    assert not offenders, "Unresolved relative imports:\n" + "\n".join(offenders)
