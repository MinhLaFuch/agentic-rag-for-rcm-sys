```python
def test_low_quality_triggers_reflection():
    """Low recommendation quality (retrieval only, no ranking) triggers reflection when threshold is enabled."""
    # SQLTool returns candidates but no ranking step (quality = 0.5)
    tool = FakeSQLTool(fixed_data={"candidates": build_items("item1"), "total_matches": 1, "truncated": False, "applied_filters": {}})

    # Plan 1: only SQLTool (no ranking) - low quality
    plan1 = build_plan_json({"tool": "SQLTool", "args": {"filters": {"domain": "Video_Games"}}})
    # Plan 2: SQLTool + RecoModelTool (full pipeline) - high quality
    plan2 = build_plan_json(
        {"tool": "SQLTool", "args": {"filters": {"domain": "Video_Games"}}},
        {"tool": "RecoModelTool", "args": {"user_id": "u1", "candidates": "$1.candidates", "top_k": 5}},
    )

    llm = ScriptedLLM(plan1, plan2)
    # Set quality threshold to 0.7 to enable quality-based reflection
    executor = PlanExecutor(llm, [tool, FakeRecoModelTool()], reflect_rounds=1, reflect_quality_threshold=0.7)
    results = executor.run("test")

    # Should trigger reflection due to low quality
    assert executor.last_reflections == 1
    assert executor.last_first_attempt_ok is False
    assert len(executor.last_attempt_plans) == 2
```
